        // =====================================================
        // QUESTION 1: Shortest & Longest Word
        // =====================================================

        function findWords() {

            var sentence =
                document.getElementById("sentence").value.trim();

            if (sentence === "") {

                document.getElementById("wordResult").innerHTML =
                    "Please enter a sentence.";

                return;
            }

            var words = sentence.split(/\s+/);

            var shortest = words[0];
            var longest = words[0];

            for (var i = 1; i < words.length; i++) {

                if (words[i].length < shortest.length) {
                    shortest = words[i];
                }

                if (words[i].length > longest.length) {
                    longest = words[i];
                }
            }

            document.getElementById("wordResult").innerHTML =
                "Shortest: " + shortest +
                "<br>" +
                "Longest: " + longest;
        }


        // =====================================================
        // QUESTION 2: BMI Calculator
        // =====================================================

        function calculateBMI() {

            var weight =
                parseFloat(
                    document.getElementById("weight").value
                );

            var height =
                parseFloat(
                    document.getElementById("height").value
                );

            if (
                isNaN(weight) ||
                isNaN(height) ||
                weight <= 0 ||
                height <= 0
            ) {

                document.getElementById("score").innerHTML = "";

                document.getElementById("type").innerHTML =
                    "Please enter valid weight and height.";

                return;
            }

            var bmi =
                weight / (height * height);

            var bodyType;

            if (bmi < 18) {

                bodyType = "Underweight";

            } else if (bmi <= 25) {

                bodyType = "Normal";

            } else if (bmi <= 30) {

                bodyType = "Overweight";

            } else {

                bodyType = "Obese";
            }

            document.getElementById("score").innerHTML =
                bmi.toFixed(2);

            document.getElementById("type").innerHTML =
                bodyType;
        }


        // =====================================================
        // QUESTION 3: Fruit Farm
        // =====================================================

        function grow() {

            var rows =
                parseInt(
                    document.getElementById("rows").value
                );

            var columns =
                parseInt(
                    document.getElementById("columns").value
                );

            var garden =
                document.getElementById("garden");

            if (
                isNaN(rows) ||
                isNaN(columns) ||
                rows <= 0 ||
                columns <= 0
            ) {

                garden.innerHTML =
                    "Please enter valid rows and columns.";

                return;
            }

            var fruits = [
                "Orange.jpg",
                "Apple.jpg"
            ];

            garden.innerHTML = "";

            for (var i = 0; i < rows; i++) {

                for (var j = 0; j < columns; j++) {

                    var random =
                        Math.floor(
                            Math.random() * fruits.length
                        );

                    var image =
                        document.createElement("img");

                    image.src =
                        fruits[random];

                    image.width = 100;
                    image.height = 100;

                    image.alt = "Fruit";

                    garden.appendChild(image);
                }

                garden.appendChild(
                    document.createElement("br")
                );
            }
        }


        // =====================================================
        // QUESTION 4: Currency Converter
        // =====================================================

        function convert() {

            var amount =
                parseFloat(
                    document.getElementById("data").value
                );

            var from =
                document.getElementById("from").value;

            var to =
                document.getElementById("to").value;

            if (
                isNaN(amount) ||
                amount < 0
            ) {

                document.getElementById("answer").innerHTML =
                    "Please enter a valid amount.";

                return;
            }

            var result;

            if (
                from === "INR" &&
                to === "USD"
            ) {

                result = amount * 0.015;

            } else if (
                from === "USD" &&
                to === "INR"
            ) {

                result = amount * 67.12;

            } else {

                result = amount;
            }

            document.getElementById("answer").innerHTML =
                result.toFixed(2) + " " + to;
        }


        // =====================================================
        // QUESTION 5: Coin Flip
        // =====================================================

        var heads = 0;
        var tails = 0;

        document
            .getElementById("flip")
            .addEventListener(
                "click",
                flipFunction
            );


        function flipFunction() {

            var random =
                Math.floor(
                    Math.random() * 2
                );

            if (random === 0) {

                document.getElementById("coin").src =
                    "heads.jpg";

                heads++;

            } else {

                document.getElementById("coin").src =
                    "tails.jpg";

                tails++;
            }

            document.getElementById("num_heads").innerHTML =
                heads;

            document.getElementById("num_tails").innerHTML =
                tails;
        }

        // QUESTION 6: Birthday Cake Order
        // =====================================================

        document
            .getElementById("submit")
            .addEventListener(
                "click",
                calculateOrder

        function calculateOrder() {

            var small =
                parseInt(
                    document.getElementById("small").value
                ) || 0;

            var medium =
                parseInt(
                    document.getElementById("medium").value
                ) || 0;

            var large =
                parseInt(
                    document.getElementById("large").value
                ) || 0;


            // Cake cost
            var cakeCost =
                (small * 100) +
                (medium * 200) +
                (large * 400);


            // 9% tax
            var tax =
                cakeCost * 0.09;

            var totalAfterTax =
                cakeCost + tax;


            // Selected tip
            var selectedTip =
                document.querySelector(
                    'input[name="tip"]:checked'
                );

            var tipPercent =
                parseInt(
                    selectedTip.value
                );


            // Tip calculation
            var tip =
                totalAfterTax *
                (tipPercent / 100);


            // Final cost
            var total =
                totalAfterTax + tip;


            // Display result
            document.getElementById("price").innerHTML =
                total.toFixed(2);
        }
